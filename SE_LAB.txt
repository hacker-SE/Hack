# SECTION 1: INITIAL SETUP

## 1.1 Check Git Version
git --version

## 1.2 Configure Git Identity (One-Time)
git config --global user.name "Your Name"
git config --global user.email "your_email@example.com"
git config --global init.defaultBranch main

## 1.3 View All Configurations
git config --list

# SECTION 2: REPOSITORY INITIALIZATION

## 2.1 Initialize New Repository
cd /path/to/project
git init
git status

## 2.2 Add Remote Repository
git remote add origin https://github.com/<username>/<repo>.git
git remote -v

## 2.3 If Remote Already Exists
git remote set-url origin https://github.com/<username>/<repo>.git

## 2.4 Remove and Re-Add Remote
git remote remove origin
git remote add origin https://github.com/<username>/<repo>.git

## 2.5 First Commit and Push
git add .
git status
git commit -m "Initial commit"
git branch -M main
git push -u origin main

# SECTION 5: REMOTE REPOSITORY MANAGEMENT

git remote -v                          # List remotes with URLs
git remote add origin <url>            # Add new remote
git remote set-url origin <new-url>    # Change remote URL
git remote remove origin               # Remove remote
git remote rename origin upstream      # Rename remote
git remote show origin                 # Detailed remote info
git remote prune origin                # Remove outdated remote branches
git fetch origin                       # Fetch without merging
git fetch origin <branch_name>         # Fetch specific branch
git fetch upstream                     # Fetch from renamed remote
git pull origin main                   # Fetch + merge
git pull --rebase origin main          # Fetch + rebase
git push origin main                   # Push to main
git push -u origin <branch>            # Push and set upstream
git push --set-upstream origin <branch>
git push --force-with-lease            # Safe force push
git push origin --delete <branch>      # Delete remote branch
git push --tags                        # Push all tags
git push origin v1.0                   # Push specific tag

---

# SECTION 6: BRANCHING

## 6.1 Listing Branches
git branch                # Local branches
git branch -a             # All local + remote branches
git branch -r             # Remote branches only

## 6.2 Creating Branches
git branch <branch-name>              # Create only
git checkout <branch-name>            # Switch to existing
git checkout -b <branch-name>         # Create + switch
git switch <branch-name>              # Switch (newer syntax)
git switch -c <branch-name>           # Create + switch

## 6.3 Renaming & Deleting
git branch -m <old-name> <new-name>   # Rename branch
git branch -d <branch-name>           # Delete (safe, merged only)
git branch -D <branch-name>           # Force delete (unmerged)
git branch -d branch1 branch2 branch3 # Delete multiple

## 6.4 Checking Merge Status
git branch --merged                   # List merged branches
git branch --no-merged                # List unmerged branches

---

# SECTION 7: MERGING & REBASING

## 7.1 Merge
git checkout main
git merge feature/homepage
git merge --abort                     # Abort merge in progress

## 7.2 Rebase
git checkout feature/homepage
git rebase main
git rebase origin/main
git rebase --continue                 # After resolving conflicts
git rebase --abort                    # Cancel rebase
git rebase -i HEAD~3                  # Interactive rebase (3 commits)

---

# SECTION 8: VIEWING HISTORY & CHANGES

git status                            # Working tree status
git diff                              # Unstaged changes
git diff --staged                     # Staged changes
git diff --cached                     # Same as --staged
git diff <branch1> <branch2>          # Compare branches
git diff main..feature                # Compare branch differences
git log                               # Full commit history
git log --oneline                     # Condensed history
git log --oneline --graph --all       # Visual tree of all branches
git log main..feature                 # Commits in feature not in main
git log --oneline <branch1>..<branch2>
git show <commit-hash>                # Details of a commit
git show fa582a1                      # Show specific commit
git blame <file>                      # Who changed each line
git blame -L 25,25 script.py          # Who changed line 25

---

# SECTION 9: UNDOING CHANGES

## 9.1 Discard / Restore
git restore <file>                    # Discard working directory changes
git restore --staged <file>           # Unstage file (keep changes)
git checkout -- <file>                # Old syntax for discard

## 9.2 Reset
git reset <file>                      # Unstage file
git reset HEAD~1                      # Undo last commit (mixed)
git reset --soft HEAD~1               # Undo commit, keep staged
git reset --mixed HEAD~1              # Undo commit, keep unstaged
git reset --hard HEAD~1               # Undo commit, discard changes

## 9.3 Revert
git revert <commit-hash>              # Undo commit via new commit
git revert HEAD                       # Revert latest commit

## 9.4 Amend
git commit --amend -m "Corrected message"
git commit --amend --no-edit          # Amend without changing message

# SECTION 11: STASHING

git stash                             # Save uncommitted changes
git stash list                        # List all stashes
git stash apply                       # Restore (keep stash)
git stash pop                         # Restore + remove stash
git stash drop                        # Delete a stash
git stash clear                       # Delete all stashes

## Typical Stash Workflow
git stash
git switch <another-branch>
# do urgent work
git switch main
git stash apply

# SECTION 16: PATCHES

git format-patch -1 HEAD              # Create patch from last commit
git format-patch -3                   # Create patches from last 3 commits
git format-patch <base>..HEAD         # Range of commits
git apply my-changes.patch            # Apply patch
git apply --check my-changes.patch    # Verify patch applies cleanly
git am 0001-Your-commit.patch         # Apply patch with commit

# SECTION 24: KEY CONCEPTS

1. Four Areas: Working Directory → Staging Area → Local Repo → Remote Repo
2. HEAD points to the latest commit in the current branch
3. origin = default remote name
4. main = default branch name
5. Merge preserves history; Rebase creates linear history
6. Fetch downloads; Pull = Fetch + Merge
7. Reset moves pointer back; Revert creates new undo commit
8. Stash = temporary storage for uncommitted changes
9. Reflog = safety net to recover lost commits
10. .gitignore prevents unwanted files from being tracked
