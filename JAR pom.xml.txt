dockerfile

For java project

# Base image with JDK
FROM openjdk:17-jdk-slim

# Set working directory inside container
WORKDIR /app

# Copy your built JAR file (replace with actual name)
COPY target/<YOUR_JAR_FILE>.jar app.jar

# Expose the port your app runs on
EXPOSE 8080

# Run the JAR
ENTRYPOINT ["java", "-jar", "app.jar"]


For web project
# Base image with Tomcat
FROM tomcat:9.0-jdk17-openjdk

# Maintainer info (optional)
LABEL maintainer="<YOUR_NAME_OR_EMAIL>"

# Remove default ROOT app
RUN rm -rf /usr/local/tomcat/webapps/ROOT

# Copy your WAR file into Tomcat (replace with actual name)
COPY target/<YOUR_WAR_FILE>.war /usr/local/tomcat/webapps/ROOT.war

# Expose Tomcat port
EXPOSE 8080

# Start Tomcat
CMD ["catalina.sh", "run"]
